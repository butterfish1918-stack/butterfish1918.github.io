const getTrackColor = (index, luminance = 50, saturation = 80) => {
    let hue = 0;
    if (index < 4)
        hue = 0 + (index * 15);
    else if (index < 10)
        hue = 60 + ((index - 4) * 25);
    else
        hue = 180 + ((index - 10) * 25);
    const dynamicSat = Math.min(100, saturation + (index * 1.5));
    return `hsla(${hue}, ${dynamicSat}%, ${luminance}%, 1)`;
};
